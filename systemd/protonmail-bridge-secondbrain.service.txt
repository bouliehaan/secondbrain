[Unit]
Description=Proton Mail Bridge for MagicMirror Second Brain
After=network-online.target <REDACTED_EMAIL>
Wants=network-online.target
Requires=<REDACTED_EMAIL>

[Service]
Type=simple

User=calendar-display
Group=calendar-display

WorkingDirectory=/home/calendar-display

Environment=HOME=/home/calendar-display
Environment=XDG_CONFIG_HOME=/home/calendar-display/.config
Environment=XDG_CACHE_HOME=/home/calendar-display/.cache
Environment=XDG_DATA_HOME=/home/calendar-display/.local/share
Environment=GNUPGHOME=/home/calendar-display/.gnupg
